cmake_minimum_required(VERSION 3.24)
if(POLICY CMP0207)
    cmake_policy(SET CMP0207 NEW)
endif()
project(Charlita VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Widgets Multimedia Svg)
if(UNIX AND NOT APPLE)
    find_package(Qt6 REQUIRED COMPONENTS WaylandClient)
    set(QT_DEPLOY_PLUGINS_DIR plugins)
    set(QT_DEPLOY_QML_DIR qml)
endif()
qt_standard_project_setup(REQUIRES 6.8)
find_program(CARGO_EXECUTABLE cargo REQUIRED)
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(RUST_PROFILE debug)
    set(RUST_ARGS)
else()
    set(RUST_PROFILE release)
    set(RUST_ARGS --release)
endif()
if(WIN32)
    set(RUST_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/target/${RUST_PROFILE}/charlita.lib")
else()
    set(RUST_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/target/${RUST_PROFILE}/libcharlita.a")
endif()
file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS "src/*.rs" "overlay/*")
add_custom_command(OUTPUT "${RUST_LIBRARY}"
    COMMAND "${CARGO_EXECUTABLE}" build --lib ${RUST_ARGS} --locked
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
    DEPENDS ${RUST_SOURCES} Cargo.toml Cargo.lock
    COMMENT "Building the Rust engine" VERBATIM)
add_custom_target(charlita_rust DEPENDS "${RUST_LIBRARY}")
function(charlita_ui_target target module main_source)
qt_add_executable(${target} ${main_source} ui/bridge.cpp ui/bridge.h)
add_dependencies(${target} charlita_rust)
target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/ui")
target_compile_definitions(${target} PRIVATE CHARLITA_QML_URI="${module}")
set_source_files_properties(ui/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(${target} URI ${module} VERSION 1.0
    QML_FILES ui/Main.qml ui/Theme.qml ui/ActionButton.qml ui/Check.qml ui/Field.qml ui/FocusScroll.qml ui/Avatar.qml ui/Scene.qml ui/Library.qml ui/Settings.qml ui/GuestEditor.qml ui/CharacterEditor.qml
    RESOURCES ui/logo.svg ui/fonts/InterVariable.ttf)
target_link_libraries(${target} PRIVATE "${RUST_LIBRARY}" Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Widgets Qt6::Multimedia Qt6::Svg)
if(WIN32)
    target_link_libraries(${target} PRIVATE userenv ws2_32 bcrypt ntdll advapi32 crypt32 secur32 shell32 ole32 runtimeobject winmm dxgi d3d11)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(DBUS REQUIRED IMPORTED_TARGET dbus-1)
    find_package(X11 REQUIRED)
    target_link_libraries(${target} PRIVATE dl pthread m rt util X11::X11 PkgConfig::DBUS)
    target_link_libraries(${target} PRIVATE Qt6::WaylandClient)
endif()
endfunction()
charlita_ui_target(charlita Charlita ui/main.cpp)
if(WIN32)
    set_target_properties(charlita PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    charlita_ui_target(charlita-ui-tests CharlitaTests ui/tests.cpp)
    target_link_libraries(charlita-ui-tests PRIVATE Qt6::Test)
    add_test(NAME native-ui COMMAND charlita-ui-tests)
    set_tests_properties(native-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 90)
endif()
install(TARGETS charlita RUNTIME DESTINATION bin)
set(CHARLITA_DEPLOY_OPTIONS)
if(Qt6_VERSION VERSION_GREATER_EQUAL 6.10)
    # System Qt installations can contain unrelated desktop image plugins.
    # Include only the Qt codecs used by Charlita and desktop platform plugins.
    list(APPEND CHARLITA_DEPLOY_OPTIONS EXCLUDE_PLUGIN_TYPES imageformats generic egldeviceintegrations platforminputcontexts
        INCLUDE_PLUGINS qjpeg qwebp qgif qsvg qsvgicon qoffscreen qwayland qwayland-egl qwayland-generic composeplatforminputcontextplugin ibusplatforminputcontextplugin)
endif()
if(UNIX AND NOT APPLE)
    list(APPEND CHARLITA_DEPLOY_OPTIONS PRE_EXCLUDE_REGEXES "^(ld-linux.*|lib(c|m|pthread|dl|rt|util|nss_.*|resolv)[.]so.*)$")
    if(Qt6_VERSION VERSION_LESS 6.10)
        # Older deploy helpers select X11 by default. Official SDK plugins have
        # a relative runpath compatible with this standard install layout.
        get_target_property(qt_qmake Qt6::qmake IMPORTED_LOCATION)
        execute_process(COMMAND "${qt_qmake}" -query QT_INSTALL_PLUGINS
            OUTPUT_VARIABLE qt_plugin_source OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
        file(GLOB extra_platforms "${qt_plugin_source}/platforms/libqoffscreen.so" "${qt_plugin_source}/platforms/libqwayland*.so")
        install(FILES ${extra_platforms} DESTINATION plugins/platforms)
    endif()
endif()
qt_generate_deploy_qml_app_script(TARGET charlita OUTPUT_SCRIPT deploy_script ${CHARLITA_DEPLOY_OPTIONS})
install(SCRIPT ${deploy_script})
